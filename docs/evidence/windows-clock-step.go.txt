package main

import (
	"fmt"
	"sort"
	"time"
)

func main() {
	hist := map[time.Duration]int{}
	start := time.Now()
	last := start
	n := 0
	for time.Since(start) < 2*time.Second {
		now := time.Now()
		n++
		if d := now.Sub(last); d > 0 {
			hist[d.Round(10*time.Microsecond)]++
			last = now
		}
	}
	keys := make([]time.Duration, 0, len(hist))
	for k := range hist {
		keys = append(keys, k)
	}
	sort.Slice(keys, func(i, j int) bool { return hist[keys[i]] > hist[keys[j]] })
	fmt.Println("reads", n, "distinct advances", len(keys))
	for i, k := range keys {
		if i == 12 {
			break
		}
		fmt.Println(k, hist[k])
	}
}
